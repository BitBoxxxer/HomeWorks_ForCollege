## 1. Эвакуация людей от огня к выходу (анимация)

```csharp
using System;
using System.Collections.Generic;
using System.Threading;

class Person
{
    public double X, Y;
    public bool Saved;
    public Person(double x, double y) { X = x; Y = y; Saved = false; }
}

class Fire
{
    public double X, Y;
    public Fire(double x, double y) { X = x; Y = y; }
}

class Program
{
    static void Main()
    {
        const int WIDTH = 50;
        const int HEIGHT = 20;
        const int EXIT_X = WIDTH - 2;
        const int EXIT_Y = HEIGHT / 2;
        const double SPEED = 1.2;
        const double FIRE_SPEED = 0.6;
        const double DT = 0.2;

        Random rand = new Random();
        List<Person> people = new List<Person>();
        for (int i = 0; i < 15; i++)
            people.Add(new Person(rand.Next(5, WIDTH - 10), rand.Next(3, HEIGHT - 3)));

        Fire fire = new Fire(WIDTH / 2, HEIGHT / 2);

        int step = 0;
        while (true)
        {
            step++;
            // ----- Движение людей к выходу -----
            foreach (var p in people)
            {
                if (p.Saved) continue;
                double dx = EXIT_X - p.X;
                double dy = EXIT_Y - p.Y;
                double len = Math.Sqrt(dx * dx + dy * dy);
                if (len > 0.1)
                {
                    p.X += (dx / len) * SPEED * DT;
                    p.Y += (dy / len) * SPEED * DT;
                }
                // Проверка достижения выхода
                if (Math.Abs(p.X - EXIT_X) < 1.0 && Math.Abs(p.Y - EXIT_Y) < 1.0)
                    p.Saved = true;
            }

            // ----- Движение огня к ближайшему человеку -----
            Person closest = null;
            double minDist = double.MaxValue;
            foreach (var p in people)
            {
                if (p.Saved) continue;
                double d = Math.Sqrt((fire.X - p.X) * (fire.X - p.X) + (fire.Y - p.Y) * (fire.Y - p.Y));
                if (d < minDist) { minDist = d; closest = p; }
            }
            if (closest != null)
            {
                double dx = closest.X - fire.X;
                double dy = closest.Y - fire.Y;
                double len = Math.Sqrt(dx * dx + dy * dy);
                if (len > 0.1)
                {
                    fire.X += (dx / len) * FIRE_SPEED * DT;
                    fire.Y += (dy / len) * FIRE_SPEED * DT;
                }
            }

            // ----- Отрисовка -----
            Console.Clear();
            for (int y = 0; y < HEIGHT; y++)
            {
                for (int x = 0; x < WIDTH; x++)
                {
                    // Стены
                    if (x == 0 || x == WIDTH - 1 || y == 0 || y == HEIGHT - 1)
                        Console.Write('#');
                    // Выход
                    else if (x == EXIT_X && y == EXIT_Y)
                    {
                        Console.ForegroundColor = ConsoleColor.Green;
                        Console.Write('E');
                        Console.ResetColor();
                    }
                    // Огонь
                    else if (Math.Abs(x - fire.X) < 0.8 && Math.Abs(y - fire.Y) < 0.8)
                    {
                        Console.ForegroundColor = ConsoleColor.Red;
                        Console.Write('F');
                        Console.ResetColor();
                    }
                    else
                    {
                        bool hasPerson = false;
                        foreach (var p in people)
                            if (!p.Saved && Math.Abs(p.X - x) < 0.6 && Math.Abs(p.Y - y) < 0.6)
                            { hasPerson = true; break; }
                        if (hasPerson)
                        {
                            Console.ForegroundColor = ConsoleColor.Cyan;
                            Console.Write('H');
                            Console.ResetColor();
                        }
                        else Console.Write('.');
                    }
                }
                Console.WriteLine();
            }
            Console.WriteLine($"Шаг {step} | Спасено: {people.FindAll(p => p.Saved).Count} | Огонь: ({fire.X:F1},{fire.Y:F1})");

            // Проверка окончания: все спасены или кто-то сгорел
            bool allSaved = true;
            foreach (var p in people)
                if (!p.Saved) { allSaved = false; break; }
            if (allSaved) { Console.WriteLine("Все спаслись!"); break; }

            bool someoneBurned = false;
            foreach (var p in people)
                if (!p.Saved && Math.Sqrt((p.X - fire.X) * (p.X - fire.X) + (p.Y - fire.Y) * (p.Y - fire.Y)) < 1.2)
                { someoneBurned = true; break; }
            if (someoneBurned) { Console.WriteLine("Кто-то сгорел!"); break; }

            Thread.Sleep(80);
        }
        Console.ReadKey();
    }
}
```

## 2. Лесной пожар (клеточный автомат) – анимация в консоли

```csharp
using System;
using System.Threading;

class ForestFire
{
    static void Main()
    {
        int size = 30;
        int EMPTY = 0, TREE = 1, FIRE = 2;
        int[,] grid = new int[size, size];
        Random rand = new Random();

        // Заполняем деревьями
        for (int i = 0; i < size; i++)
            for (int j = 0; j < size; j++)
                grid[i, j] = rand.NextDouble() < 0.6 ? TREE : EMPTY;

        // Поджигаем центр
        grid[size / 2, size / 2] = FIRE;

        double pGrowth = 0.02;   // рост новых деревьев
        double pLightning = 0.005; // молния

        for (int step = 0; step < 200; step++)
        {
            int[,] next = new int[size, size];
            for (int i = 0; i < size; i++)
                for (int j = 0; j < size; j++)
                {
                    if (grid[i, j] == FIRE)
                        next[i, j] = EMPTY; // огонь гаснет
                    else if (grid[i, j] == EMPTY)
                        next[i, j] = rand.NextDouble() < pGrowth ? TREE : EMPTY;
                    else // TREE
                    {
                        bool burningNear = false;
                        for (int di = -1; di <= 1; di++)
                            for (int dj = -1; dj <= 1; dj++)
                            {
                                int ni = i + di, nj = j + dj;
                                if (ni >= 0 && ni < size && nj >= 0 && nj < size && grid[ni, nj] == FIRE)
                                    burningNear = true;
                            }
                        if (burningNear || rand.NextDouble() < pLightning)
                            next[i, j] = FIRE;
                        else
                            next[i, j] = TREE;
                    }
                }
            grid = next;

            // Отрисовка
            Console.Clear();
            for (int i = 0; i < size; i++)
            {
                for (int j = 0; j < size; j++)
                {
                    if (grid[i, j] == EMPTY) Console.Write(' ');
                    else if (grid[i, j] == TREE)
                    {
                        Console.ForegroundColor = ConsoleColor.Green;
                        Console.Write('T');
                    }
                    else if (grid[i, j] == FIRE)
                    {
                        Console.ForegroundColor = ConsoleColor.Red;
                        Console.Write('F');
                    }
                    Console.ResetColor();
                }
                Console.WriteLine();
            }
            Console.WriteLine($"Шаг {step}");
            Thread.Sleep(100);
        }
    }
}
```

## 3. Модель трафика (кольцевая дорога) – анимация в консоли

```csharp
using System;
using System.Threading;

class Traffic
{
    static void Main()
    {
        int roadLen = 60;
        int Vmax = 5;
        double density = 0.3;
        double probSlow = 0.3;
        Random rand = new Random();

        int[] road = new int[roadLen];
        for (int i = 0; i < roadLen; i++) road[i] = -1;
        int numCars = (int)(roadLen * density);
        int placed = 0;
        while (placed < numCars)
        {
            int pos = rand.Next(roadLen);
            if (road[pos] == -1) { road[pos] = 0; placed++; }
        }

        for (int time = 0; time < 200; time++)
        {
            int[] newRoad = new int[roadLen];
            for (int i = 0; i < roadLen; i++) newRoad[i] = -1;

            for (int i = 0; i < roadLen; i++)
            {
                if (road[i] == -1) continue;
                int v = road[i];
                // Ускорение
                if (v < Vmax) v++;
                // Дистанция до впереди идущей машины
                int d = 1;
                while (road[(i + d) % roadLen] == -1 && d < roadLen) d++;
                if (v >= d) v = d - 1;
                // Случайное замедление
                if (v > 0 && rand.NextDouble() < probSlow) v--;
                // Перемещение
                int newPos = (i + v) % roadLen;
                newRoad[newPos] = v;
            }
            road = newRoad;

            // Отрисовка
            Console.Clear();
            for (int i = 0; i < roadLen; i++)
            {
                if (road[i] == -1) Console.Write('.');
                else
                {
                    Console.ForegroundColor = ConsoleColor.Yellow;
                    Console.Write(road[i]);
                    Console.ResetColor();
                }
            }
            Console.WriteLine($"  Время {time}");
            Thread.Sleep(100);
        }
    }
}
```

## 4. Эпидемия (агенты движутся случайно, заражают друг друга)

```csharp
using System;
using System.Collections.Generic;
using System.Threading;

class Person
{
    public double X, Y, Vx, Vy;
    public bool Infected;
    public Person(Random rnd, bool infected)
    {
        X = rnd.NextDouble() * 78 + 1;
        Y = rnd.NextDouble() * 23 + 1;
        Vx = (rnd.NextDouble() - 0.5) * 2;
        Vy = (rnd.NextDouble() - 0.5) * 2;
        Infected = infected;
    }
    public void Move(double dt)
    {
        X += Vx * dt;
        Y += Vy * dt;
        if (X <= 0) { X = 0; Vx = -Vx; }
        if (X >= 79) { X = 79; Vx = -Vx; }
        if (Y <= 0) { Y = 0; Vy = -Vy; }
        if (Y >= 24) { Y = 24; Vy = -Vy; }
    }
}

class Epidemic
{
    static void Main()
    {
        const int WIDTH = 80;
        const int HEIGHT = 25;
        const int NUM_PEOPLE = 40;
        const double DT = 0.2;
        const double INFECT_RADIUS = 2.5;
        Random rand = new Random();

        List<Person> people = new List<Person>();
        for (int i = 0; i < NUM_PEOPLE - 1; i++) people.Add(new Person(rand, false));
        people.Add(new Person(rand, true)); // нулевой пациент

        for (int step = 0; step < 500; step++)
        {
            // Движение
            foreach (var p in people) p.Move(DT);
            // Заражение
            for (int i = 0; i < people.Count; i++)
                for (int j = i + 1; j < people.Count; j++)
                {
                    double dx = people[i].X - people[j].X;
                    double dy = people[i].Y - people[j].Y;
                    if (Math.Sqrt(dx * dx + dy * dy) < INFECT_RADIUS)
                        if (people[i].Infected || people[j].Infected)
                            people[i].Infected = people[j].Infected = true;
                }

            // Отрисовка
            Console.Clear();
            for (int y = 0; y < HEIGHT; y++)
            {
                for (int x = 0; x < WIDTH; x++)
                {
                    bool hasPerson = false;
                    bool infected = false;
                    foreach (var p in people)
                        if (Math.Abs(p.X - x) < 0.5 && Math.Abs(p.Y - y) < 0.5)
                        { hasPerson = true; infected = p.Infected; break; }
                    if (hasPerson)
                    {
                        Console.ForegroundColor = infected ? ConsoleColor.Red : ConsoleColor.Green;
                        Console.Write('O');
                        Console.ResetColor();
                    }
                    else Console.Write('.');
                }
                Console.WriteLine();
            }
            int infectedCount = people.FindAll(p => p.Infected).Count;
            Console.WriteLine($"Шаг {step} | Заражено: {infectedCount}/{NUM_PEOPLE}");
            Thread.Sleep(50);
            if (infectedCount == NUM_PEOPLE) break;
        }
    }
}
```

## 5. Поиск парковочного места (агенты, анимация)

```csharp
using System;
using System.Collections.Generic;
using System.Threading;

class ParkingSlot
{
    public int X, Y;
    public bool Occupied;
    public ParkingSlot(int x, int y) { X = x; Y = y; Occupied = false; }
}

class Car
{
    public double X, Y;
    public ParkingSlot Target;
    public bool Parked;
    public Car(double x, double y) { X = x; Y = y; Target = null; Parked = false; }
}

class ParkingSim
{
    static void Main()
    {
        const int WIDTH = 80;
        const int HEIGHT = 25;
        const int NUM_CARS = 12;
        const double SPEED = 1.2;
        const double DT = 0.2;
        const double SCAN_RADIUS = 12.0;
        Random rand = new Random();

        // Парковочные места (вверху и внизу)
        List<ParkingSlot> slots = new List<ParkingSlot>();
        for (int x = 10; x <= 70; x += 8)
        {
            slots.Add(new ParkingSlot(x, 3));
            slots.Add(new ParkingSlot(x, HEIGHT - 4));
        }

        List<Car> cars = new List<Car>();
        for (int i = 0; i < NUM_CARS; i++)
            cars.Add(new Car(2, HEIGHT / 2)); // въезд слева по центру

        int step = 0;
        while (true)
        {
            step++;
            // Логика для каждой машины
            foreach (var car in cars)
            {
                if (car.Parked) continue;

                // Если нет цели – ищем свободное место в радиусе
                if (car.Target == null)
                {
                    foreach (var slot in slots)
                    {
                        if (slot.Occupied) continue;
                        double dx = slot.X - car.X;
                        double dy = slot.Y - car.Y;
                        double dist = Math.Sqrt(dx * dx + dy * dy);
                        if (dist < SCAN_RADIUS)
                        {
                            car.Target = slot;
                            slot.Occupied = true;
                            break;
                        }
                    }
                }

                // Движение к цели или просто вперёд
                if (car.Target != null)
                {
                    double dx = car.Target.X - car.X;
                    double dy = car.Target.Y - car.Y;
                    double len = Math.Sqrt(dx * dx + dy * dy);
                    if (len > 0.8)
                    {
                        car.X += (dx / len) * SPEED * DT;
                        car.Y += (dy / len) * SPEED * DT;
                    }
                    else
                    {
                        car.Parked = true;
                    }
                }
                else
                {
                    // Едем вправо
                    car.X += SPEED * DT;
                    if (car.X > WIDTH - 5) car.X = WIDTH - 5;
                }
            }

            // Отрисовка
            Console.Clear();
            // Рисуем парковочные места
            for (int y = 0; y < HEIGHT; y++)
            {
                for (int x = 0; x < WIDTH; x++)
                {
                    bool isSlot = false;
                    bool occupied = false;
                    foreach (var s in slots)
                        if (Math.Abs(s.X - x) < 1 && Math.Abs(s.Y - y) < 1)
                        { isSlot = true; occupied = s.Occupied; break; }
                    if (isSlot)
                    {
                        Console.ForegroundColor = occupied ? ConsoleColor.Red : ConsoleColor.Green;
                        Console.Write('S');
                        Console.ResetColor();
                    }
                    else
                    {
                        bool isCar = false;
                        foreach (var c in cars)
                            if (!c.Parked && Math.Abs(c.X - x) < 0.6 && Math.Abs(c.Y - y) < 0.6)
                            { isCar = true; break; }
                        if (isCar)
                        {
                            Console.ForegroundColor = ConsoleColor.Cyan;
                            Console.Write('C');
                            Console.ResetColor();
                        }
                        else Console.Write('.');
                    }
                }
                Console.WriteLine();
            }
            int parked = cars.FindAll(c => c.Parked).Count;
            Console.WriteLine($"Шаг {step} | Припарковано: {parked}/{NUM_CARS}");
            Thread.Sleep(80);
            if (parked == NUM_CARS) break;
        }
    }
}
```

```csharp
using System;
class Logistic
{
    static void Main()
    {
        double N = 10, K = 500, r = 0.5, dt = 0.2, t = 0;
        while (t < 30) { N += r * N * (1 - N / K) * dt; t += dt; }
        Console.WriteLine($"Через 30 ед. времени N = {Math.Round(N)}");
    }
}
```

### Монте-Карло: вероятность встречи

```csharp
using System;
class MonteCarloMeeting
{
    static void Main()
    {
        Random rnd = new Random();
        int total = 100000, meet = 0;
        for (int i = 0; i < total; i++)
        {
            double t1 = rnd.NextDouble() * 60;
            double t2 = rnd.NextDouble() * 60;
            if (Math.Abs(t1 - t2) <= 10) meet++;
        }
        Console.WriteLine($"Вероятность встречи: {100.0 * meet / total:F2}%");
    }
}
```
